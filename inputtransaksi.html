// Ubah loadExistingPayments menjadi: async function loadExistingPayments() { try { // Coba dengan filter let transaksiSnap; try { transaksiSnap = await db.collection('transaksi') .where('jenis_iuran', '==', jenis) .where('tipe', '==', 'pemasukan') .get(); } catch(e) { // Jika gagal (mungkin karena indeks), ambil semua dan filter manual transaksiSnap = await db.collection('transaksi').get(); } const map = new Map(); transaksiSnap.forEach(doc => { const data = doc.data(); // Jika filter gagal, kita filter manual const dataJenis = data.jenis_iuran || 'konsumsi'; if (dataJenis !== jenis) return; if (data.tipe !== 'pemasukan' && data.tipe !== 'Pemasukan') return; if (data.santriId && data.bulanIdx !== undefined) { if (!map.has(data.santriId)) map.set(data.santriId, new Set()); map.get(data.santriId).add(data.bulanIdx); } }); console.log('Payment map size:', map.size); return map; } catch(err) { console.error(err); return new Map(); } }
